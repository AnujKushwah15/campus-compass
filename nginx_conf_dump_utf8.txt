# configuration file /etc/nginx/nginx.conf:
events {}

http {
    limit_req_zone $binary_remote_addr zone=api:10m rate=30r/m;
    
    server {
        server_name thanganat25.com www.thanganat25.com;
    
        # ΓöÇΓöÇ Campus Compass Backend API (port 3001) ΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇ
        location /api/ {
            proxy_pass http://127.0.0.1:3001/api/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
    
        # ΓöÇΓöÇ MediaMTX stream auth callback ΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇ
        location /stream-auth {
            proxy_pass http://127.0.0.1:3001/stream-auth;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
    
        # ΓöÇΓöÇ MediaMTX WebRTC stream (port 8889) ΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇΓöÇ
        location /stream/ {
            # Read token from query args, fallback to cookie
            set $stream_token $arg_token;
            if ($stream_token = "") {
                set $stream_token $cookie_stream_token;
            }
    
            # If it came from the query, save it in a Cookie for subsequent XHR requests!
            if ($arg_token != "") {
                add_header Set-Cookie "stream_token=$arg_token; Path=/stream/; HttpOnly; SameSite=None; Secure" always;
            }
    
            # Append token to args if it was missing 
            set $modified_args $args;
            if ($arg_token = "") {
                set $modified_args "$args&token=$stream_token";
            }

    
            rewrite ^/stream/(.*) /$1 break;
            proxy_pass http://127.0.0.1:8889$uri?$modified_args;
    
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
    
            # WebRTC needs WebSocket support
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
    
            # Longer timeouts for WebRTC negotiation
            proxy_read_timeout 86400;
            proxy_send_timeout 86400;
            
            # Add broad CORS here too as WebRTC sometimes preflights
            add_header Access-Control-Allow-Origin * always;
        }
    
        listen 443 ssl; # managed by Certbot
        ssl_certificate /etc/letsencrypt/live/thanganat25.com/fullchain.pem; # managed by Certbot
        ssl_certificate_key /etc/letsencrypt/live/thanganat25.com/privkey.pem; # managed by Certbot
        include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
        ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
    }
    
    server {
        if ($host = www.thanganat25.com) {
            return 301 https://$host$request_uri;
        }
    
        if ($host = thanganat25.com) {
            return 301 https://$host$request_uri;
        }
    
        listen 80;
        server_name thanganat25.com www.thanganat25.com;
        return 404;
    }
}
